const screen_w: u32 = {{width}};
const screen_h: u32 = {{height}};
const screen_title = "{{title}}";
const target_fps: u32 = {{fps}};

const BackendGfxType = @import("backend_gfx");

{{module_vars}}pub fn main() !void {
    var gpa = std.heap.DebugAllocator(.{}).init;
    defer _ = gpa.deinit();
    const allocator = gpa.allocator();

{{hidden_setup}}    window.initWindow(screen_w, screen_h, screen_title);
    defer window.closeWindow();
    window.setTargetFPS(target_fps);

{{hooks_init_block}}
    var g = AssembledGame.init(allocator);
    defer g.deinit();

    g.setHooks(&hooks);

    g.setScreenHeight(@as(f32, @floatFromInt(screen_h)));

{{preview_setup}}{{setup_code}}
    // Screenshot capture request (labelle-cli#227). Reads
    // `LABELLE_SCREENSHOT_PATH` once on startup; null when the env var is
    // absent, so the per-frame check below collapses to a `null` test on
    // regular runs. `loop_start_ns` captures wall-clock at the first frame
    // so `after_sec` is "seconds since the game entered its main loop".
    var screenshot_req: ?engine.ScreenshotRequest = engine.requestedScreenshot();
    const loop_start_ns: i128 = engine.nowNs();

    while (!window.windowShouldClose()) {
        const dt: f32 = 0.016;
        // Apply a pending fullscreen/windowed switch requested via
        // `game.setFullscreen`/`toggleFullscreen`. The engine owns the
        // desired flag; the backend owns the platform call. Gated on BOTH
        // the backend exposing `setFullscreen` and the engine exposing the
        // drain, so the template stays compatible with backends/engines
        // that predate the API (older example projects still build).
        if (comptime @hasDecl(window, "setFullscreen") and @hasDecl(@TypeOf(g), "takeFullscreenRequest")) {
            if (g.takeFullscreenRequest()) |on| window.setFullscreen(on);
        }
        // Aspect-fit the design canvas (config) into the real window size,
        // so content renders correctly even if the window differs from the
        // configured default. Per-frame is cheap and survives resizes.
        BackendGfxType.setScreenSize(window.getScreenWidth(), window.getScreenHeight());
        BackendGfxType.setDesignSize(@intCast(screen_w), @intCast(screen_h));
{{preview_heartbeat}}{{tick_code}}        g.tick(dt);

        window.beginDrawing();
        window.clearBackground(245, 245, 245, 255);
        g.render();
        g.renderGizmos();
{{gui_draw_code}}        window.drawText(screen_title, 10, 10, 20, 80, 80, 80, 255);

        // Queue the screenshot exactly once, after the configured delay,
        // BEFORE `endDrawing` so the capture lands on THIS frame's content
        // (scene + GUI overlay just submitted). bgfx's `takeScreenshot`
        // only stashes the path; `endDrawing` issues the bgfx capture
        // request right before the frame swap and pumps the extra frames
        // the multi-threaded render path needs to flush the async capture
        // to disk. Gated on the backend exposing the API so the template
        // stays compatible with older bgfx backends.
        //
        // We intentionally do NOT break out of the loop after the capture:
        // unlike raylib's synchronous `TakeScreenshot`, bgfx's clean exit
        // would run `gpa.deinit()`, and any pre-existing teardown leak in
        // the engine/plugins surfaces as a panic under the game's debug
        // allocator. Letting `--timeout` SIGTERM the process avoids that
        // path; the `.tga` is already on disk by the time the timeout
        // fires. (One-shot is preserved via `screenshot_req = null`.)
        if (comptime @hasDecl(window, "takeScreenshot")) {
            if (screenshot_req) |req| {
                const now_ns: i128 = engine.nowNs();
                const elapsed_sec: f32 = @as(f32, @floatFromInt(now_ns - loop_start_ns)) / 1_000_000_000.0;
                if (elapsed_sec >= req.after_sec) {
                    window.takeScreenshot(req.path);
                    screenshot_req = null;
                }
            }
        }

        window.endDrawing();
    }
}
